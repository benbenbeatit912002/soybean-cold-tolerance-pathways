### Wilcoxon rank-sum test
run_wilcox_test = function(group1, group2) {
  test_result <- wilcox.test(group1, group2, alternative = "greater", conf.level = 0.95, exact = FALSE)
  return(data.frame(
    statistic = test_result$statistic,
    p_value = test_result$p.value
  ))
}

run_t_test = function(group1, group2) {
  test_result =t.test(group1, group2, alternative = "greater", conf.level = 0.95)
  return(data.frame(
    statistic = test_result$statistic,
    p_value = test_result$p.value
  ))
}
